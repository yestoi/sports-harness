**Goal:** Put the owner's fun tickets inside the five surfaces that already exist — this week's
ideas as draft slips built on a schedule, prop legs with live stat lines, an in-app "I placed
this" and a closed correction table on a LAN listener behind an owner login, a Floor game detail
that tells the decision story — without moving a single paper number, venue write, stake,
anchor, threshold or gate criterion.

**Architecture:** Nineteen units in six slices. Slice A fixes the contracts: the policy file and
its loader (`harness/parlay/`), the Odds API prop request plus the normalizer's internal
`prop:<stat>` keys and validated links (`harness/feeds/odds_api.py`,
`harness/normalize/odds.py`), the ESPN summary/roster/game-log parsers and the player identity
map (`harness/feeds/espn.py`, `harness/normalize/players.py`), and an evidence task that records
DraftKings' settlement rules verbatim. Slice B is data and accounting: one additive Alembic
revision mirrored in the models and `harness/db/schema.py`; prop grading in
`harness/parlay/needs.py` and `harness/settlement/parlay_grade.py`; the builder's prop pool and
same-game assembly in `harness/parlay/build.py`; the scheduled `parlay_build` settle stage; three
new recorder sources (props, player stats, draft reprice); and placement/correction semantics in
`harness/parlay/placement.py`. Slice C is projections and writes: an `harness/dashboard/auth.py`
that fails closed, the LAN app factory, the two owner POST routes, and the Ticket and Floor
builders. Slice D is the interface on fixtures: tokens, the Ticket surface with its confirm
sheet, the Floor game detail. Slices E/F are the second serve container behind a compose profile,
the first live NFL window measurement, and the verification rows.

**Tech Stack:** Python 3.12, FastAPI + uvicorn (TLS through stdlib `ssl`), Jinja2,
SQLAlchemy 2 (ORM models plus Core `text()` statements), Alembic, Typer, pytest, PostgreSQL 16,

```python
def valid_dk_link(url: str | None) -> str | None:
    """An `https` DraftKings link of at most 300 characters, or None.

    Byte for byte: the query string carries the venue's own outcome id, so the string is
    returned as it arrived rather than re-assembled from parts.

    if not isinstance(url, str) or not url or len(url) > LINK_MAX:

    try:
        parts = urlsplit(url)
    except ValueError:
```

```python
class OddsPropSnapshot(Base):
    """One DraftKings player-prop outcome (addendum §3.3 and §9 as amended; D23).

    Its own table rather than four columns on `odds_snapshots`: a prop outcome is keyed by the
    **player**, and `uq_odds_snapshot_row` keys on `coalesce(outcome_team_id, -1)` with no
    `where` clause, so two scorers in one `prop:anytime_td` market are one key to it. Making
    them fit would mean rebuilding a unique index on a bulk table, which is not additive; a new
    table is. Column spellings are copied from `OddsSnapshot`.
    """
    __tablename__ = "odds_prop_snapshots"
    id: Mapped[int] = mapped_column(BigInteger, primary_key=True, autoincrement=True)
    raw_id: Mapped[int] = mapped_column(BigInteger, nullable=False)
    book: Mapped[str] = mapped_column(String(32), nullable=False)
    game_id: Mapped[int | None] = mapped_column(Integer)
    market_type: Mapped[str] = mapped_column(String(24), nullable=False)
    player_name: Mapped[str] = mapped_column(String(80), nullable=False)
    #: Null until Task 3's resolver fills it; `ix_odds_prop_lookup` is partial on it.
    player_id: Mapped[int | None] = mapped_column(Integer)
    outcome_side: Mapped[str | None] = mapped_column(String(8))
    point: Mapped[Decimal | None] = mapped_column(Numeric(6, 1))
    #: The only price column, as on `OddsSnapshot`: the venue is asked for decimal odds and
    #: `pricing.american()` derives the American price for a prop exactly as for a game line.
    price_decimal: Mapped[Decimal] = mapped_column(Numeric(10, 4), nullable=False)
    book_last_update: Mapped[datetime | None] = mapped_column(DateTime(timezone=True))
    fetched_at: Mapped[datetime] = mapped_column(DateTime(timezone=True), nullable=False)
    link: Mapped[str | None] = mapped_column(String(300))
    sid: Mapped[str | None] = mapped_column(String(64))
    __table_args__ = (
        # The prop upsert's conflict target: one row per (fetch, book, market, player, side,
        # line). `player_name` is the column `uq_odds_snapshot_row` could not carry.
        Index("uq_odds_prop_row", "raw_id", "book", "market_type", "player_name",

        # orders. Partial, so a row whose player never resolved costs nothing.
```

```python
@dataclass(frozen=True)
class StatState:
    """One player's newest recorded value for one stat (addendum §4.3, §4.4).

    `source_ts` is null for ESPN's summary -- it carries no per-stat timestamp -- so the surface
    shows the fetch age instead. `final` is the game's, not the stat's: a value read before the
    final box score is provisional, because a later poll may correct it downward.
    """
    stat: str
    value: Decimal
    source_ts: datetime | None
    final: bool


def _whole(line: Decimal) -> bool:
    return line == line.to_integral_value()


def _prop_needs(leg: LegSpec, stat: StatState | None) -> str:
    if stat is None:
        return "no stat yet"
    if leg.operator == "yes":
        return "scored · provisional until final" if stat.value > 0 else "no touchdown yet"
    line = leg.threshold
    if line is None:
        return "no rule for this bet"
    if leg.operator in ("over", "atleast"):
        if stat.value < line:
            return f"{_points(line - stat.value)} to go"
        return (f"reached {_points(stat.value)} of {_points(line)} · provisional until final")
    #: under
    if stat.value < line:
        return f"under by {_points(line - stat.value)} · live until the game ends"
    return f"over by {_points(stat.value - line)}"


def _prop_outcome(leg: LegSpec, stat: StatState | None) -> str | None:
    """The final grade of one prop leg, or None while there is nothing to grade.

    `None` on a missing `StatState` and on a non-final one: absence is not a miss (§4.4). A
    whole-number line pushes on `over` and `under` and never on `atleast` -- `225+` is a
    different market, not the same market read differently.
    """
    if stat is None or not stat.final:
        return None
    if leg.operator == "yes":
        return "hit" if stat.value > 0 else "miss"
    line = leg.threshold
    if line is None:
        raise ValueError("prop leg has no threshold")
    if leg.operator == "atleast":
        return "hit" if stat.value >= line else "miss"
    if leg.operator == "over":
        if stat.value > line:
            return "hit"
        return "push" if (stat.value == line and _whole(line)) else "miss"
    if leg.operator == "under":
        if stat.value < line:
            return "hit"
        return "push" if (stat.value == line and _whole(line)) else "miss"
```

```python
def newest_dk_prop_price(session: Session, game_id: int, market_type: str, player_id: int,
                         point: Decimal | None, side: str | None, now: datetime,
```

```python
SLOTS = tuple((sport, shape) for sport in ("nfl", "ncaaf")
              for shape in ("smart", "lottery", "lottery_same_game"))
#: America/Chicago weekday and hour a slot's build becomes due (§2.4).
BUILD_TIMES = {"ncaaf": (4, 18), "nfl": (5, 18)}      # Friday, Saturday
#: The stage stops *starting* builds once it has spent this much of the settler's shared budget
#: (§2.4). A build already running finishes; the next run picks the slot up.
STAGE_BUDGET_S = 60
#: B-I9: the one model call a build makes is bounded, and a timeout falls back to the template.
RATIONALE_TIMEOUT_S = 30


def run_parlay_build(session: Session, now: datetime, budget: Budget) -> StageResult:
    """The stage. The signature is fixed at `(session, now, budget)` by `job.py`'s `StageFn`, so
    settings come from `get_settings()` here rather than from a parameter, and the clock is the
    budget's own -- `Budget` holds the `monotonic` the job injected, which a test can pin, and
    this module reads no clock of its own.
    """
    settings = get_settings()
    counts = {"built": 0, "replaced": 0, "expired": 0, "refused": 0, "errors": 0}
    counts["expired"] = expire_cards(session, now)
    year, week = chicago_iso_week(now)
    #: Yield when the stage has spent its own allowance out of the job's shared budget, the
    #: `parlay_grade.MIN_BUDGET_S` pattern: one slot must never run the job past its period.
    floor_s = budget.remaining_s() - STAGE_BUDGET_S
    exhausted = False
    for sport, shape in SLOTS:
        if not budget.ok() or budget.remaining_s() <= floor_s:
            exhausted = True
            break
        _run_slot(session, settings, sport, shape, year, week, now, counts)
    return StageResult(name="parlay_build", counts=counts, budget_exhausted=exhausted)


register_stage("parlay_build", run_parlay_build)
```

`_run_slot` is where the rationale timeout is applied: it calls `build_card(session, settings,
sport, week, kind, now, year=year, same_game=(shape == "lottery_same_game"),
parent_card_id=parent, timeout_s=RATIONALE_TIMEOUT_S)`, and `build_card` passes `timeout_s`
straight through to `write_rationale` (Task 6 adds both keywords). The constant lives here and
travels as an argument, so nothing under `harness/parlay/` ever imports the settlement package.

Each slot, in `SLOTS` order: skip when a `proposed`, `placed` or `alive` card already exists for
`(year, week, sport, shape)`; skip with `not_built_yet` before the build time; refuse with
`week_at_cap` when `week_staked(session, year, week) >= config.weekly_budget`; otherwise call
`build_card(..., same_game=shape == "lottery_same_game")` inside `try/except BuildRefused` and

```python
SCRYPT_N, SCRYPT_R, SCRYPT_P = 16384, 8, 1
SCRYPT_MAXMEM = 64 * 1024 * 1024
COOKIE_NAME = "sports_session"
MAX_AGE_DAYS = 180
LOGIN_FAILURES_PER_MINUTE = 5


def hash_password(password: str) -> str:
    salt = secrets.token_bytes(16)
    digest = hashlib.scrypt(password.encode(), salt=salt, n=SCRYPT_N, r=SCRYPT_R, p=SCRYPT_P,
                            maxmem=SCRYPT_MAXMEM)
    return (f"scrypt${SCRYPT_N}${SCRYPT_R}${SCRYPT_P}$"
            f"{base64.b64encode(salt).decode()}${base64.b64encode(digest).decode()}")


def verify_password(password: str, line: str) -> bool:
    """Constant-time comparison against a line whose parameters are pinned.

    A line carrying other parameters raises `ValueError` rather than verifying against them: it
    is a hash someone else's tool wrote, and accepting it would let a weaker cost factor decide
    who reaches the write routes. The exception message names the parameter, never the line.
    """
```

```python
    def parlay_placed(body: PlacedBody, _=Depends(write_rules)) -> dict:
        """Record a slip the owner placed by hand at DraftKings (§5.1).

        This route records; it never places. Every refusal `mark_placed` makes is kept, and the

            if existing is not None:
                return {"placement": _as_json(existing)}
            try:

                                        clock(), body.leg_lines,
                                        confirmation_id=body.confirmation_id, note=body.note)
                s.commit()

                raise _refusal(409, "confirmation_reused")

                               placement=_as_json(_placement_of(s, body.card_id)))
            except BudgetExceeded as exc:

                               left=str(exc.left))
            except LineMoved as exc:
                raise _refusal(409, "line_moved",
                               moved={str(k): [str(a), str(b)] for k, (a, b) in exc.legs.items()})
            except IntegrityError:
                s.rollback()
                return {"placement": _as_json(_placement_of(s, body.card_id))}
            return {"placement": _as_json(placement)}
```

```python
def _luminance(hex_colour: str) -> float:
    """WCAG 2.1 relative luminance. Written out rather than imported: the point of the check is
    that it is independent of whatever the stylesheet says about itself."""
    channels = [int(hex_colour.lstrip("#")[i:i + 2], 16) / 255 for i in (0, 2, 4)]
    linear = [c / 12.92 if c <= 0.03928 else ((c + 0.055) / 1.055) ** 2.4 for c in channels]
    return 0.2126 * linear[0] + 0.7152 * linear[1] + 0.0722 * linear[2]


def _contrast(a: str, b: str) -> float:
    la, lb = _luminance(a), _luminance(b)
    return (max(la, lb) + 0.05) / (min(la, lb) + 0.05)
```

```javascript
//: The only writer, in the only module allowed to reach the network. Sends the CSRF header and
//: the JSON content type the routes require, and reports a 401 as the session-required state so
//: the shell can show the login prompt rather than a silent failure.
export async function postJson(path, body) {
  try {
    const response = await fetch(path, {
      method: "POST",
      headers: { "Content-Type": "application/json", "X-Requested-With": "sports-ui" },
      credentials: "same-origin",
      body: JSON.stringify(body),
    });
    const payload = response.headers.get("Content-Type")?.includes("json")
      ? await response.json() : null;
    return { status: response.status, body: payload };
  } catch (error) {
    return { status: 0, body: null };
  }
}
```

```
/srv/sports-harness/sports-compose exec -T postgres psql -X -v ON_ERROR_STOP=1 -U harness -d harness -At <<'SQL'
-- 1. Update cadence: the gap between consecutive rows for one carded player's stat.
select player_id, stat, count(*) as rows,
       round(avg(extract(epoch from ts - lag(ts) over w))) as mean_gap_s,
       max(extract(epoch from ts - lag(ts) over w)) as max_gap_s
from player_stat_events window w as (partition by player_id, stat order by ts)
where ts >= now() - interval '8 hours' group by player_id, stat order by rows desc limit 20;
-- 2. Corrections: how many rows lowered a value, and by how much.
select count(*) filter (where correction) as corrections, count(*) as rows
from player_stat_events where ts >= now() - interval '8 hours';
-- 3. Missing players: carded legs with no row at all after their game went final.
select l.id, l.player_id, l.stat from parlay_legs l join parlay_cards c on c.id = l.card_id
join games g on g.id = l.game_id
where c.status in ('placed','alive') and g.status like 'final%' and l.market_type = 'prop'
  and not exists (select 1 from player_stat_events e
                  where e.game_id = l.game_id and e.player_id = l.player_id and e.stat = l.stat);
-- 4. Prop credits for the month against the allocation.
select key, credits_used, last_fetched_at from source_state where key like 'odds_props:%';
SQL
```
