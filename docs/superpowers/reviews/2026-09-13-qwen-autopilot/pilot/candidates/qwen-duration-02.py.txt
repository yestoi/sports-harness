"""Parse ASCII duration strings like '1h30m' into integer milliseconds."""

import re

_MAX_MS = 31536000000
_MAX_LEN = 128

# Magnitudes in strictly descending order; rank encodes that order.
_UNIT_FACTOR = {'d': 86400000, 'h': 3600000, 'm': 60000, 's': 1000, 'ms': 1}
_UNIT_RANK = {'d': 0, 'h': 1, 'm': 2, 's': 3, 'ms': 4}

# One token: ASCII digits (leading zeroes allowed) immediately followed by a
# unit. 'ms' is listed before 'm' so a token like '1ms' is consumed whole
# (longest unit matching). Only literal ASCII spaces may pad the ends or
# separate tokens; at least one token is required.
_TOKEN = r'[0-9]+(?:ms|d|h|m|s)'
_FULL = re.compile(r' *(' + _TOKEN + r')( *(' + _TOKEN + r'))* *$')
_FIND = re.compile(r'([0-9]+)(ms|d|h|m|s)')


def parse_duration(text):
    if not isinstance(text, str):
        raise TypeError('parse_duration expects a string, got %r' % type(text).__name__)
    if len(text) > _MAX_LEN:
        raise ValueError('duration longer than %d characters' % _MAX_LEN)
    if _FULL.fullmatch(text) is None:
        raise ValueError('invalid duration: %r' % text)
    total = 0
    prev_rank = -1
    for amount_s, unit in _FIND.findall(text):
        rank = _UNIT_RANK[unit]
        # Strictly descending magnitude, no unit repetition.
        if rank <= prev_rank:
            raise ValueError('units must appear in strictly descending '
                             'magnitude: %r' % text)
        prev_rank = rank
        total += int(amount_s) * _UNIT_FACTOR[unit]
    if total > _MAX_MS:
        raise ValueError('duration exceeds %d ms' % _MAX_MS)
    return total
