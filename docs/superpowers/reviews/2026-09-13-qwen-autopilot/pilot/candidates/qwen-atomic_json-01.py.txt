"""Atomic local JSON snapshot writer."""

import json
import os
import tempfile
from pathlib import Path


def write_snapshot(path, value):
    """Atomically write the JSON snapshot of ``value`` to ``path``.

    The value is serialized exactly as
    ``json.dumps(value, ensure_ascii=False, sort_keys=True, indent=2,
    allow_nan=False) + '\\n'`` encoded as UTF-8 and written via a
    uniquely named temporary regular file in the destination's parent,
    which is then atomically replaced at ``path`` with ``os.replace``.

    On serialization or filesystem failure, the original exception is
    propagated, any pre-existing destination contents are preserved, and
    only the temporary file created by this call is removed. Parent
    directories are never created.
    """
    destination = Path(path)
    payload = (
        json.dumps(
            value, ensure_ascii=False, sort_keys=True, indent=2, allow_nan=False
        )
        + '\n'
    ).encode('utf-8')

    fd = None
    temp_path = None
    try:
        fd, temp_path = tempfile.mkstemp(
            prefix=destination.name + '.', suffix='.tmp', dir=str(destination.parent)
        )
        with os.fdopen(fd, 'wb') as handle:
            fd = None  # Descriptor ownership transfers to the file object.
            handle.write(payload)
            handle.flush()
        os.replace(temp_path, destination)
        temp_path = None
    finally:
        if fd is not None:
            os.close(fd)
        if temp_path is not None:
            try:
                os.unlink(temp_path)
            except OSError:
                pass
    return None
